
import pandas as pd
from sklearn.preprocessing import LabelEncoder
from sklearn.tree import DecisionTreeClassifier, plot_tree
import matplotlib.pyplot as plt

# Load dataset
df = pd.read_csv("/content/sample_data/cars (1).csv")

X = df.iloc[:, :-1].copy()
y = df.iloc[:, -1]

# Encode data
le = LabelEncoder()
for col in X.columns:
    X[col] = LabelEncoder().fit_transform(X[col])

y = le.fit_transform(y)

# ID3
id3 = DecisionTreeClassifier(criterion="entropy")
id3.fit(X, y)

# CART
cart = DecisionTreeClassifier(criterion="gini")
cart.fit(X, y)

# Input
print("Enter car details:")
user = []
for col in X.columns:
    val = input(col + ": ")
    user.append(LabelEncoder().fit(df[col]).transform([val])[0])

# Prediction
print("ID3 Prediction:", le.inverse_transform(id3.predict([user]))[0])
print("CART Prediction:", le.inverse_transform(cart.predict([user]))[0])

# Plot trees
plt.figure(figsize=(12, 5))
plot_tree(id3, feature_names=X.columns, filled=True)
plt.title("ID3 Decision Tree")
plt.show()

plt.figure(figsize=(12, 5))
plot_tree(cart, feature_names=X.columns, filled=True)
plt.title("CART Decision Tree")
plt.show()
