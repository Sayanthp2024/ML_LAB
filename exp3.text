
import pandas as pd
import numpy as np

# Load dataset
df = pd.read_csv("/content/sample_data/Titanic-Dataset (1).csv")

# Select columns
df = df[["Survived", "Pclass", "Sex", "Age", "SibSp", "Parch", "Fare"]]
df["Sex"] = (df["Sex"] == "female").astype(int)
df["Age"] = df["Age"].fillna(df["Age"].mean())
df = df.fillna(0)

X = df.iloc[:, 1:].values
y = df["Survived"].values

# Normalize data
mean, std = X.mean(0), X.std(0) + 1e-8
X = (X - mean) / std
X = np.c_[np.ones(len(X)), X]

# Split dataset
np.random.seed(42)
idx = np.random.permutation(len(X))
n = int(0.8 * len(X))
Xt, Xv = X[idx[:n]], X[idx[n:]]
yt, yv = y[idx[:n]], y[idx[n:]]

# Logistic Regression
w = np.zeros(Xt.shape[1])
lr = 0.01

def sigmoid(z):
    return 1 / (1 + np.exp(-np.clip(z, -500, 500)))

for i in range(1000):
    p = sigmoid(Xt @ w)
    w -= lr * Xt.T @ (p - yt) / len(yt)

# Accuracy
pred = (sigmoid(Xv @ w) >= 0.5).astype(int)
print("Accuracy:", round(np.mean(pred == yv) * 100, 2), "%")

# User input
pc = int(input("Passenger Class (1/2/3): "))
sex = int(input("Sex (0=Male, 1=Female): "))
age = float(input("Age: "))
sib = int(input("Siblings/Spouses: "))
par = int(input("Parents/Children: "))
fare = float(input("Fare: "))

# Predict survival
u = np.array([[pc, sex, age, sib, par, fare]])
u = (u - mean) / std
u = np.c_[np.ones(1), u]

prob = sigmoid(u @ w)[0]
print("Survival Probability:", round(prob * 100, 2), "%")
print("Prediction:", "SURVIVED" if prob >= 0.5 else "NOT SURVIVED")
